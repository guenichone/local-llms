from functools import lru_cache


@lru_cache(maxsize=None)
def fibonacci(n: int) -> int:
    """Return the nth Fibonacci number using memoization.

    The Fibonacci sequence is defined as:
        F(0) = 0
        F(1) = 1
        F(n) = F(n-1) + F(n-2) for n >= 2

    Parameters
    ----------
    n : int
        The position in the Fibonacci sequence (0-indexed).

    Returns
    -------
    int
        The Fibonacci number at position ``n``.

    Raises
    ------
    ValueError
        If ``n`` is negative.

    Examples
    --------
    >>> fibonacci(0)
    0
    >>> fibonacci(1)
    1
    >>> fibonacci(10)
    55
    """
    if n < 0:
        raise ValueError(f"n must be a non-negative integer, got {n}")
    if n == 0:
        return 0
    if n == 1:
        return 1
    return fibonacci(n - 1) + fibonacci(n - 2)


def fibonacci_sequence(count: int) -> list[int]:
    """Return the first ``count`` Fibonacci numbers.

    Parameters
    ----------
    count : int
        Number of Fibonacci numbers to generate. Must be non-negative.

    Returns
    -------
    list[int]
        A list of the first ``count`` Fibonacci numbers, starting from F(0).

    Raises
    ------
    ValueError
        If ``count`` is negative.

    Examples
    --------
    >>> fibonacci_sequence(5)
    [0, 1, 1, 2, 3]
    >>> fibonacci_sequence(0)
    []
    """
    if count < 0:
        raise ValueError(f"count must be non-negative, got {count}")
    return [fibonacci(i) for i in range(count)]


# ── Demonstration ──────────────────────────────────────────────────────────

if __name__ == "__main__":
    print("Fibonacci sequence (first 15 numbers):")
    for i, val in enumerate(fibonacci_sequence(15)):
        print(f"  F({i:2d}) = {val}")

    print("\nRandom lookups:")
    for n in (0, 1, 2, 5, 10, 20, 50):
        print(f"  F({n:2d}) = {fibonacci(n)}")
