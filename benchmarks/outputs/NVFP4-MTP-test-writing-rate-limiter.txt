class TestConcurrentAccess:
    """Test thread safety and concurrent access."""

    def test_concurrent_consume_no_overconsumption(self, bucket):
        """Verify concurrent consumers don't exceed capacity."""
        num_threads = 20
        results = []
        lock = threading.Lock()

        def consume_one():
            result = bucket.consume(1)
            with lock:
                results.append(result)

        threads = [threading.Thread(target=consume_one) for _ in range(num_threads)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()

        # At most 5 should succeed (capacity=5)
        successful = sum(1 for r in results if r)
        assert successful <= bucket.capacity

    def test_concurrent_consume_all_succeed_within_capacity(self, bucket):
        """Verify exactly capacity number of concurrent consumes succeed."""
        num_threads = 10
        results = []
        lock = threading.Lock()

        def consume_one():
            result = bucket.consume(1)
            with lock:
                results.append(result)

        threads = [threading.Thread(target=consume_one) for _ in range(num_threads)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()

        successful = sum(1 for r in results if r)
        assert successful == bucket.capacity

    def test_concurrent_refill_and_consume(self, slow_bucket):
        """Verify concurrent refill and consume operations are safe."""
        num_consumers = 10
        results = []
        lock = threading.Lock()

        def consume_one():
            result = bucket.consume(1)
            with lock:
                results.append(result)

        def refill():
            time.sleep(0.5)  # 5 tokens at rate=0.1
            bucket.consume(1)

        threads = [threading.Thread(target=consume_one) for _ in range(num_consumers)]
        threads.append(threading.Thread(target=refill))
        for t in threads:
            t.start()
        for t in threads:
            t.join()

        # All consumers should succeed since refill replenishes tokens
        successful = sum(1 for r in results if r)
        assert successful == num_consumers

    def test_concurrent_no_deadlock(self, bucket):
        """Verify no deadlock occurs under concurrent access."""
        num_threads = 100
        results = []
        lock = threading.Lock()

        def consume_one():
            result = bucket.consume(1)
            with lock:
                results.append(result)

        threads = [threading.Thread(target=consume_one) for _ in range(num_threads)]
        start = time.monotonic()
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        elapsed = time.monotonic() - start

        assert elapsed < 5.0  # Should complete quickly, no deadlock
        assert len(results) == num_threads
