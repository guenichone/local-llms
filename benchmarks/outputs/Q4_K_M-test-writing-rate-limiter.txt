class TestBasicConsumption:

        assert bucket.tokens == 8

        assert bucket.tokens == 0

        assert bucket.tokens == 0

    def test_consume_returns_true_when_sufficient(self, bucket):
        bucket.consume(5)
        assert bucket.consume(3) is True

class TestRefillTiming:

    def test_refill_after_time_passes(self, bucket):
        bucket.consume(10)
        assert bucket.tokens == 0
        time.sleep(1.2)
        assert bucket.tokens >= 1.0, f"Expected >= 1.0, got {bucket.tokens}"

    def test_refill_rate(self, bucket):
        bucket.consume(10)
        time.sleep(0.5)
        # rate=1.0, elapsed ~0.5s -> tokens ~0.5
        assert 0.4 <= bucket.tokens <= 0.6, f"Expected ~0.5, got {bucket.tokens}"

    def test_refill_does_not_exceed_capacity(self, bucket):
        bucket.consume(10)
        time.sleep(5)  # rate=1.0, elapsed ~5s -> would be 5.0 but capped at 10
        assert bucket.tokens <= 10.0

    def test_refill_partial(self, bucket):
        bucket.consume(8)
        time.sleep(0.3)
        # Should have refilled ~0.3 tokens
        assert bucket.tokens >= 0.2 and bucket.tokens <= 0.5

    def test_refill_immediate(self, bucket):
        # No time passes between consuming and checking
        bucket.consume(10)
        assert bucket.tokens == 0
        assert bucket.tokens + 0.0 >= 0  # should still be 0

class TestConcurrency:

    def test_concurrent_consumption_exact_tokens(self):
        """10 threads, 10 tokens -> all should succeed."""
        bucket = TokenBucket(rate=1000.0, capacity=10)
        bucket.consume(10)
        assert bucket.tokens == 0

        results = []
        lock = threading.Lock()

        def consumer():
            ok = bucket.consume(1)
            with lock:
                results.append(ok)

        with ThreadPoolExecutor(max_workers=10) as executor:
            futures = [executor.submit(consumer) for _ in range(10)]
            for future in as_completed(futures):
                future.result()  # raise any exceptions

        assert len(results) == 10
        assert all(results), "All 10 threads should have succeeded"
        assert bucket.tokens == 0

    def test_concurrent_consumption_exceeds_tokens(self):
        """11 threads, 10 tokens -> only 10 should succeed."""
        bucket = TokenBucket(rate=1000.0, capacity=10)
        bucket.consume(10)
        assert bucket.tokens == 0

        results = []
        lock = threading.Lock()

        def consumer():
            ok = bucket.consume(1)
            with lock:
                results.append(ok)

        with ThreadPoolExecutor(max_workers=11) as executor:
            futures = [executor.submit(consumer) for _ in range(11)]
            for future in as_completed(futures):
                future.result()

        assert len(results) == 11
        assert sum(results) == 10, f"Expected 10 successes, got {sum(results)}"
        assert bucket.tokens == 0

    def test_concurrent_refill_during_consumption(self):
        """Concurrent threads while tokens are being refilled."""
        bucket = TokenBucket(rate=100.0, capacity=10)
        bucket.consume(5)

        results = []
        lock = threading.Lock()

        def consumer():
            ok = bucket.consume(1)
            with lock:
                results.append(ok)

        with ThreadPoolExecutor(max_workers=5) as executor:
            futures = [executor.submit(consumer) for _ in range(5)]
            for future in as_completed(futures):
                future.result()

        assert sum(results) == 5, f"Expected 5 successes, got {sum(results)}"
        # After refill, tokens should be <= capacity
        assert bucket.tokens <= 10.0

    def test_concurrent_high_frequency(self):
        """Stress test with many threads and rapid consumption."""
        bucket = TokenBucket(rate=10000.0, capacity=100)
        bucket.consume(100)

        num_threads = 100
        num_tokens = 100
        results = []
        lock = threading.Lock()

        def consumer():

            with lock:
                results.append(ok)

        with ThreadPoolExecutor(max_workers=num_threads) as executor:
            futures = [executor.submit(consumer) for _ in range(num_threads)]
            for future in as_completed(futures):
                future.result()

        assert sum(results) == num_tokens, f"Expected {num_tokens} successes, got {sum(results)}"
        assert bucket.tokens == 0

    def test_concurrent_with_refill(self):
        """Test concurrent access while tokens are actively refilling."""
        bucket = TokenBucket(rate=100.0, capacity=5)
        bucket.consume(5)

        results = []
        lock = threading.Lock()

        def consumer():
            ok = bucket.consume(1)
            with lock:
                results.append(ok)

        with ThreadPoolExecutor(max_workers=5) as executor:
            futures = [executor.submit(consumer) for _ in range(5)]
            for future in as_completed(futures):
                future.result()

        assert sum(results) == 5

    def test_concurrent_zero_tokens(self):
        """Multiple threads try to consume zero tokens."""
        bucket = TokenBucket(rate=1.0, capacity=10)
        assert bucket.consume(0) is True

        results = []
        lock = threading.Lock()

        def consumer():
            ok = bucket.consume(0)

        assert bucket.tokens == 10

        num_threads = 200
