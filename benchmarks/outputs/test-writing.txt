import pytest
from parse_log_line import parse_log_line


class TestValidLogLines:
    """Test cases for correctly formatted log lines."""

    def test_valid_line_with_ip_address(self):
        """Test a full log line with IP address (docstring example)."""
        line = "[ERROR] 2024-01-15 10:30:00 - user123 - Failed login from 192.168.1.1"
        result = parse_log_line(line)
        assert result is not None
        assert result['level'] == 'ERROR'
        assert result['timestamp'] == '2024-01-15 10:30:00'
        assert result['user_id'] == 'user123'
        assert result['message'] == 'Failed login'
        assert result['ip_address'] == '192.168.1.1'

    def test_valid_line_without_ip_address(self):
        """Test a log line without an IP address."""
        line = "[INFO] 2024-01-15 10:30:00 - admin - User logged in"
        result = parse_log_line(line)
        assert result is not None
        assert result['level'] == 'INFO'
        assert result['timestamp'] == '2024-01-15 10:30:00'
        assert result['user_id'] == 'admin'
        assert result['message'] == 'User logged in'
        assert result['ip_address'] is None

    def test_valid_line_with_different_log_levels(self):
        """Test various log levels."""
        for level in ['DEBUG', 'INFO', 'WARNING', 'ERROR', 'CRITICAL', 'TRACE']:
            line = f"[{level}] 2024-06-01 12:00:00 - testuser - Test message"
            result = parse_log_line(line)
            assert result is not None
            assert result['level'] == level

    def test_valid_line_with_different_timestamps(self):
        """Test various timestamp formats."""
        timestamps = [
            "2024-01-01 00:00:00",
            "2024-12-31 23:59:59",
            "2024-02-29 12:00:00",  # Leap year
            "2023-06-15 08:30:45",
        ]
        for ts in timestamps:
            line = f"[INFO] {ts} - user1 - Some event"
            result = parse_log_line(line)
            assert result is not None
            assert result['timestamp'] == ts

    def test_valid_line_with_message_containing_spaces(self):
        """Test a message with multiple words."""
        line = "[WARNING] 2024-03-10 15:45:00 - user456 - Disk space low on server"
        result = parse_log_line(line)
        assert result is not None
        assert result['message'] == 'Disk space low on server'

    def test_valid_line_with_message_containing_special_chars(self):
        """Test a message with special characters like punctuation."""
        line = "[ERROR] 2024-04-05 09:15:30 - user789 - Connection timeout: 30s"
        result = parse_log_line(line)
        assert result is not None
        assert result['message'] == 'Connection timeout: 30s'

    def test_valid_line_with_message_containing_from(self):
        """Test a message that contains 'from' but no IP (should not match IP)."""
        line = "[INFO] 2024-05-12 11:00:00 - user999 - Sent from email"
        result = parse_log_line(line)
        assert result is not None
        assert result['message'] == 'Sent from email'
        assert result['ip_address'] is None

    def test_valid_line_with_leading_whitespace(self):
        """Test a line with leading whitespace (should be stripped)."""
        line = "  [INFO] 2024-07-20 14:30:00 - user111 - Leading whitespace test  "
        result = parse_log_line(line)
        assert result is not None
        assert result['level'] == 'INFO'
        assert result['message'] == 'Leading whitespace test'

    def test_valid_line_with_trailing_whitespace(self):
        """Test a line with trailing whitespace."""
        line = "[DEBUG] 2024-08-15 16:00:00 - user222 - Trailing whitespace test "
        result = parse_log_line(line)
        assert result is not None
        assert result['level'] == 'DEBUG'
        assert result['message'] == 'Trailing whitespace test'

    def test_valid_line_with_ip_address_containing_zeros(self):
        """Test IP address with leading zeros (e.g., 192.168.001.001)."""
        line = "[INFO] 2024-09-10 10:00:00 - user333 - IP with leading zeros from 192.168.001.001"
        result = parse_log_line(line)
        assert result is not None
        assert result['ip_address'] == '192.168.001.001'
