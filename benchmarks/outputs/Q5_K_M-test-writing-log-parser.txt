# tests/test_parse_log_line.py
"""Comprehensive pytest tests for parse_log_line."""

import pytest

from parse_log_line import parse_log_line


# ---------------------------------------------------------------------------
# Valid log lines (all fields including IP address)
# ---------------------------------------------------------------------------

class TestValidLines:
    """parse_log_line should return a populated dict for well-formed log lines."""

    def test_docstring_example(self):
        """The exact example from the function docstring."""
        line = "[ERROR] 2024-01-15 10:30:00 - user123 - Failed login from 192.168.1.1"
        result = parse_log_line(line)
        assert result is not None
        assert result == {
            "level": "ERROR",
            "timestamp": "2024-01-15 10:30:00",
            "user_id": "user123",
            "message": "Failed login",
            "ip_address": "192.168.1.1",
        }

    def test_all_log_levels(self):
        """All common log levels should be parsed correctly."""
        for level in ("DEBUG", "INFO", "WARNING", "ERROR", "CRITICAL"):
            line = f"[{level}] 2024-06-01 12:00:00 - alice - Test message from 10.0.0.1"
            result = parse_log_line(line)
            assert result is not None
            assert result["level"] == level
            assert result["user_id"] == "alice"
            assert result["ip_address"] == "10.0.0.1"

    def test_message_with_special_characters(self):
        """Messages containing regex-sensitive characters should be handled."""
        line = "[INFO] 2024-03-10 08:15:00 - bob - Hello, world! (test 123) from 172.16.0.5"
        result = parse_log_line(line)
        assert result is not None
        assert result["message"] == "Hello, world! (test 123)"

    def test_message_with_underscores_and_dots(self):
        """User IDs and messages may contain underscores and dots."""
        line = "[WARNING] 2024-05-20 14:30:00 - user_123.456 - Reset password from 8.8.8.8"
        result = parse_log_line(line)
        assert result is not None
        assert result["user_id"] == "user_123.456"
        assert result["ip_address"] == "8.8.8.8"

    def test_message_with_escaped_brackets(self):
        """Messages containing brackets should not break parsing."""
        line = "[ERROR] 2024-01-01 00:00:00 - sys - [Stack trace] from 192.168.0.1"
        result = parse_log_line(line)
        assert result is not None
        assert result["message"] == "[Stack trace]"

    def test_message_with_multiple_spaces_before_from(self):
        """Extra whitespace before 'from' should be tolerated."""
        line = "[INFO] 2024-02-14 09:00:00 - carol   - Sent email from 10.10.10.10"
        result = parse_log_line(line)
        assert result is not None
        assert result["message"] == "Sent email"
        assert result["ip_address"] == "10.10.10.10"

    def test_message_with_colons_and_semicolons(self):
        """Punctuation-heavy messages should parse correctly."""
        line = "[DEBUG] 2024-07-04 23:59:59 - dev - x:y;z from 1.2.3.4"
        result = parse_log_line(line)
        assert result is not None
        assert result["message"] == "x:y;z"

class TestInvalidLines:
    """parse_log_line should return None for malformed or invalid log lines."""

    def test_empty_string(self):
        """An empty string should not parse."""
        assert parse_log_line("") is None

    def test_none_input(self):
        """None should not crash the function."""
        assert parse_log_line(None) is None

    def test_random_text(self):
        """Random text unrelated to log format should not parse."""
        assert parse_log_line("Hello, world!") is None

    def test_missing_brackets(self):
        """Missing surrounding brackets should fail."""
        assert parse_log_line("ERROR 2024-01-15 10:30:00 - user123 - Failed login from 192.168.1.1") is None

    def test_invalid_level(self):
        """An invalid log level should cause parsing to fail."""
        assert parse_log_line("[INVALID] 2024-01-15 10:30:00 - user123 - message from 192.168.1.1") is None

    def test_invalid_timestamp(self):
        """A timestamp that doesn't match the expected format should fail."""
        assert parse_log_line("[ERROR] 2024/01/15 10:30:00 - user123 - message from 192.168.1.1") is None

    def test_invalid_user_id(self):
        """A user ID containing spaces should fail."""
        assert parse_log_line("[ERROR] 2024-01-15 10:30:00 - user 123 - message from 192.168.1.1") is None

    def test_invalid_ip_format(self):
        """An IP address with invalid octets should fail."""
        assert parse_log_line("[ERROR] 2024-01-15 10:30:00 - user123 - message from 999.999.999.999") is None

    def test_ip_with_non_numeric_octets(self):
        """IP octets containing letters should fail."""
        assert parse_log_line("[ERROR] 2024-01-15 10:30:00 - user123 - message from 192.168.1.a") is None

    def test_line_with_extra_fields_after_ip(self):
        """Extra content after the IP address should fail."""
        assert parse_log_line("[ERROR] 2024-01-15 10:30:00 - user123 - message from 192.168.1.1 extra") is None

    def test_line_with_tabs_instead_of_spaces(self):
        """Tabs instead of spaces should fail."""
        assert parse_log_line("[ERROR]\t2024-01-15 10:30:00\t-\tuser123\t-\tmessage from 192.168.1.1") is None

    def test_line_with_leading_whitespace(self):
        """Leading whitespace should be stripped, not treated as invalid."""
        line = "  [ERROR] 2024-01-15 10:30:00 - user123 - message from 192.168.1.1  "
        result = parse_log_line(line)
        assert result is not None
        assert result["level"] == "ERROR"

    def test_line_with_trailing_whitespace(self):
        """Trailing whitespace should be stripped."""
        line = "[ERROR] 2024-01-15 10:30:00 - user123 - message from 192.168.1.1   "
        result = parse_log_line(line)
        assert result is not None
        assert result["ip_address"] == "192.168.1.1"
