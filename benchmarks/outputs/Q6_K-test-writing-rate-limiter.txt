# rate_limiter.py

import time
import threading

class TokenBucket:
    def __init__(self, rate: float, capacity: int):
        self.rate = rate
        self.capacity = capacity
        self.tokens = capacity
        self.last_refill = time.monotonic()
        self._lock = threading.Lock()
    
    def consume(self, tokens: int = 1) -> bool:
        with self._lock:
            now = time.monotonic()
            elapsed = now - self.last_refill
            self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
            self.last_refill = now
            if self.tokens >= tokens:
                self.tokens -= tokens
                return True
            return False

class TestRefillTiming:
    """Tests for token refill timing and rate accuracy."""

    def test_refill_after_wait(self, slow_bucket):
        """Tokens refill proportionally to elapsed time."""
        slow_bucket.consume(1)
        elapsed = 1.0  # seconds
        time.sleep(elapsed)
        # Expected tokens: 10 - 1 + 0.1 * 1.0 = 9.1, clamped to 10
        expected = min(slow_bucket.capacity, slow_bucket.tokens + elapsed * slow_bucket.rate)
        assert slow_bucket.tokens >= expected - 0.01
        assert slow_bucket.tokens <= expected + 0.01

    def test_refill_rate_accuracy(self, slow_bucket):
        """Refill rate is accurate within tolerance."""
        slow_bucket.consume(10)  # drain completely
        elapsed = 5.0  # seconds
        time.sleep(elapsed)
        expected = min(slow_bucket.capacity, elapsed * slow_bucket.rate)
        assert slow_bucket.tokens >= expected - 0.01
        assert slow_bucket.tokens <= expected + 0.01

    def test_no_refill_without_wait(self, bucket):
        """Tokens do not refill instantly."""
        bucket.consume(5)  # drain completely
        assert bucket.consume(1) is False
        assert bucket.tokens < bucket.capacity

    def test_refill_is_continuous(self, slow_bucket):
        """Tokens refill gradually, not in a single burst."""
        slow_bucket.consume(10)
        initial = slow_bucket.tokens
        time.sleep(0.1)
        assert slow_bucket.tokens > initial
        # Should not have reached capacity in 0.1s
        assert slow_bucket.tokens < slow_bucket.capacity - 0.5

# ─── Concurrency Tests ─────────────────────────────────────────────────────────

class TestConcurrency:
    """Tests for thread-safe operation under concurrent access."""

    def test_concurrent_consume(self):
        """Multiple threads can safely consume tokens without race conditions."""
        bucket = TokenBucket(rate=1000.0, capacity=100)
        consumed = [0]
        lock = threading.Lock()

        def worker():
            for _ in range(10):
                if bucket.consume(1):
                    with lock:
                        consumed[0] += 1

        threads = [threading.Thread(target=worker) for _ in range(10)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()

        assert consumed[0] == 100  # exactly 100 tokens consumed

    def test_concurrent_consume_no_double_count(self):
        """Concurrent consumption never exceeds total tokens."""
        bucket = TokenBucket(rate=1000.0, capacity=50)
        consumed = [0]
        lock = threading.Lock()

        def worker():
            for _ in range(10):
                if bucket.consume(1):
                    with lock:
                        consumed[0] += 1

        threads = [threading.Thread(target=worker) for _ in range(20)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()

        assert consumed[0] <= 50  # never exceed capacity

    def test_concurrent_refill_and_consume(self):
        """Refill and consume can happen concurrently without issues."""
        bucket = TokenBucket(rate=100.0, capacity=10)
        consumed = [0]
        lock = threading.Lock()

        def consumer():
            for _ in range(20):
                if bucket.consume(1):
                    with lock:
                        consumed[0] += 1

        def refiller():
            for _ in range(10):
                time.sleep(0.05)
                bucket.tokens = min(bucket.capacity, bucket.tokens + 1)

        consumer_thread = threading.Thread(target=consumer)
        refiller_thread = threading.Thread(target=refiller)

        consumer_thread.start()
        refiller_thread.start()
        consumer_thread.join()
        refiller_thread.join()

        assert consumed[0] > 0
        assert consumed[0] <= bucket.capacity * 2

    def test_concurrent_with_pool(self):
        """Concurrent access via ThreadPoolExecutor is safe."""
        bucket = TokenBucket(rate=1000.0, capacity=100)
        consumed = [0]
        lock = threading.Lock()

        def worker():
            for _ in range(5):
                if bucket.consume(1):
                    with lock:
                        consumed[0] += 1

        with ThreadPoolExecutor(max_workers=10) as executor:
            futures = [executor.submit(worker) for _ in range(20)]
            for f in as_completed(futures):
                f.result()

        assert consumed[0] == 100

    def test_concurrent_stress_test(self):
        """Stress test with many threads and high contention."""
        bucket = TokenBucket(rate=10000.0, capacity=500)
        consumed = [0]
        lock = threading.Lock()

        def worker():
            for _ in range(50):
                if bucket.consume(1):
                    with lock:
                        consumed[0] += 1

        num_threads = 50
        threads = [threading.Thread(target=worker) for _ in range(num_threads)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()

        assert consumed[0] == 500  # exactly 500 tokens consumed
        assert bucket.tokens == 0

    def test_no_deadlock(self):
        """Concurrent access does not cause deadlocks."""
        bucket = TokenBucket(rate=10.0, capacity=5)
        errors = []

        def worker():
            try:
                for _ in range(100):
                    bucket.consume(1)
            except Exception as e:
                errors.append(e)

        threads = [threading.Thread(target=worker) for _ in range(20)]
        for t in threads:
            t.start()
        for t in threads:
            t.join(timeout=10)

        assert len(errors) == 0, f"Errors encountered: {errors}"
        assert bucket.tokens >= 0

    def test_concurrent_refill_during_stress(self):
        """Refill works correctly under concurrent stress."""
        bucket = TokenBucket(rate=100.0, capacity=10)
        consumed = [0]
        lock = threading.Lock()
        refill_count = [0]
        refill_lock = threading.Lock()

        def consumer():
            for _ in range(100):
                if bucket.consume(1):
                    with lock:
                        consumed[0] += 1

        def refiller():
            for _ in range(200):
                time.sleep(0.01)
                with refill_lock:
                    bucket.tokens = min(bucket.capacity, bucket.tokens + 1)
                with refill_lock:
                    refill_count[0] += 1

        consumer_thread = threading.Thread(target=consumer)
        refiller_thread = threading.Thread(target=refiller)

        consumer_thread.start()
        refiller_thread.start()
        consumer_thread.join(timeout=30)
        refiller_thread.join(timeout=30)

        assert consumed[0] > 0
        assert refill_count[0] > 0
        assert bucket.tokens >= 0

    def test_concurrent_with_time_pressure(self):
        """Concurrent access under time pressure (many small sleeps)."""
