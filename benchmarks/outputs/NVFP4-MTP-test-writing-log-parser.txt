class TestParseEdgeCases:
    def test_return_type_is_dict(self):
        """Return type should be dict when parsing succeeds."""
        line = "[ERROR] 2024-01-15 10:30:00 - user1 - Failed login"
        result = parse_log_line(line)
        assert isinstance(result, dict)

    def test_return_type_is_none_on_failure(self):
        """Return type should be None when parsing fails."""
        assert parse_log_line("invalid") is None

    def test_required_keys_present(self):
        """Dict should contain all required keys."""
        line = "[ERROR] 2024-01-15 10:30:00 - user1 - Failed login"
        result = parse_log_line(line)
        assert result is not None
        for key in ('level', 'timestamp', 'user_id', 'message', 'ip_address'):
            assert key in result, f"Missing key: {key}"

    def test_value_types(self):
        """All values should have correct types."""
        line = "[ERROR] 2024-01-15 10:30:00 - user1 - Failed login"
        result = parse_log_line(line)
        assert isinstance(result['level'], str)
        assert isinstance(result['timestamp'], str)
        assert isinstance(result['user_id'], str)
        assert isinstance(result['message'], str)
        assert result['ip_address'] is None

    def test_level_is_uppercase(self):
        """Level should be uppercase (matching the regex \w+)."""
        line = "[error] 2024-01-15 10:30:00 - user1 - Failed login"
        result = parse_log_line(line)
        assert result is not None
        assert result['level'] == 'error'

    def test_timestamp_format(self):
        """Timestamp should match expected format."""
        line = "[ERROR] 2024-01-15 10:30:00 - user1 - Failed login"
        result = parse_log_line(line)
        assert result is not None
        assert len(result['timestamp']) == 19  # YYYY-MM-DD HH:MM:SS

    def test_message_stripped(self):
        """Message should be stripped of surrounding whitespace."""
        line = "[ERROR] 2024-01-15 10:30:00 - user1 -   Failed login   "
        result = parse_log_line(line)
        assert result is not None
        assert result['message'] == 'Failed login'

    def test_message_with_multiple_messages(self):
        """Only the first message should be captured."""
        line = "[ERROR] 2024-01-15 10:30:00 - user1 - First msg from 192.168.1.1"
        result = parse_log_line(line)
        assert result is not None
        assert result['message'] == 'First msg'
